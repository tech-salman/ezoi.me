function copyText(btn, text){ navigator.clipboard.writeText(text).then(()=>{ const old=btn.textContent; btn.textContent='Copied!'; btn.classList.add('copy-ok'); setTimeout(()=>{ btn.textContent=old; btn.classList.remove('copy-ok'); }, 1200); }).catch(()=>{ alert('Copy failed'); }); }
